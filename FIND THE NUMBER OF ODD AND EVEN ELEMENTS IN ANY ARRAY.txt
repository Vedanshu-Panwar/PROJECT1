#include<iostream>
using namespace std;
int main(){
	int r,c,a=0;
	cout<<"Enter number of rows : ";
	cin>>r;
	cout<<"Enter numbers of columns : ";
	cin>>c;
	int arr[r][c];
	cout<<"Enter elements in array : ";
	for(int i=0;i<r;i++){
		for(int j=0;j<c;j++){
			cin>>arr[i][j];
		}
	}
	for(int i=0;i<r;i++){
		for(int j=0;j<c;j++){
			if(arr[i][j]%2==0)a++;
		}
	}
	cout<<"Elements in array are : "<<endl;
	for(int i=0;i<r;i++){
		for(int j=0;j<c;j++){
			cout<<arr[i][j]<<" ";
		}
		cout<<endl;
	}
	cout<<"Number of even elements in the array are : "<<a<<endl;
	cout<<"Number of odd elements in the array are : "<<(r*c)-a<<endl;
}
